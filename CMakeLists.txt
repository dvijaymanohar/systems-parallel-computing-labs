cmake_minimum_required(VERSION 3.18)
project(systems_parallel_computing_labs LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Threads REQUIRED)
add_executable(systems_lab examples/systems_lab.cpp)
target_link_libraries(systems_lab PRIVATE Threads::Threads)
enable_testing()
add_test(NAME serial_sum COMMAND systems_lab sum 100000 1)
add_test(NAME parallel_sum COMMAND systems_lab sum 100000 4)
add_test(NAME counter COMMAND systems_lab counter 4 10000)
